\cabecalho{Gabarito 14: Simulado}{Revisão geral, no formato e no nível da Prova 3: Variáveis Aleatórias Contínuas (Cap. 4, Aulas 24--30)}

\begin{questao}{1} \textbf{(Densidade em potência)}
\begin{solucao}
\begin{enumerate}[label=(\alph*)]
  \item $\int_5^\infty Cx^{-2}\,dx = C\left[-x^{-1}\right]_5^\infty = C\left(0-(-1/5)\right) =
    C/5 = 1 \Rightarrow C=5$.
  \item $F(x) = \int_5^x 5t^{-2}\,dt = 5\left[-t^{-1}\right]_5^x = 5\left(-\frac1x+\frac15\right)
    = 1-\frac5x$, para $x\ge5$.
  \item $P(X>10) = 1-F(10) = 1-\left(1-\frac{5}{10}\right) = \frac{5}{10}=0{,}5$.
\end{enumerate}
\end{solucao}
\end{questao}

\begin{questao}{2} \textbf{(Tempo de espera: Uniforme)}
\begin{solucao}
$X\sim\text{Unif}(0,20)$.
\begin{enumerate}[label=(\alph*)]
  \item $E(X) = \dfrac{0+20}{2}=10$ minutos. $\mathrm{Var}(X) = \dfrac{(20-0)^2}{12} =
    \dfrac{400}{12}\approx 33{,}33$.
  \item $P(5\le X\le 12) = F(12)-F(5) = \dfrac{12-0}{20}-\dfrac{5-0}{20} = \dfrac{7}{20}=0{,}35$.
\end{enumerate}
\end{solucao}
\end{questao}

\begin{questao}{3} \textbf{(Normal: retorno de fundo)}
\begin{solucao}
$X\sim N(10\%,\ 18\%^2)$.
\begin{enumerate}[label=(\alph*)]
  \item $Z=\frac{15-10}{18}=0{,}278$. $P(X>15)=P(Z>0{,}278)\approx1-0{,}6094=0{,}3906$.
  \item $Z=\frac{-8-10}{18}=-1{,}0$. $P(X<-8)=P(Z<-1{,}0)\approx0{,}1587$.
\end{enumerate}
\end{solucao}
\end{questao}

\begin{questao}{4} \textbf{(Peso de pacotes)}
\begin{solucao}
\begin{enumerate}[label=(\alph*)]
  \item $\displaystyle\int_0^2 \frac38 x^2\,dx = \frac38\left[\frac{x^3}{3}\right]_0^2 =
    \frac18\left[x^3\right]_0^2 = \frac{8}{8}=1$ (\checkmark), e $f(x)\ge0$ em todo o domínio: densidade
    válida.
  \item $\displaystyle E(X) = \int_0^2 x\cdot\frac38 x^2\,dx = \frac38\int_0^2 x^3\,dx =
    \frac38\left[\frac{x^4}{4}\right]_0^2 = \frac38\times4 = 1{,}5$ kg.
  \item $\displaystyle P(X\le1) = \int_0^1 \frac38 x^2\,dx = \frac18\left[x^3\right]_0^1 =
    \frac18=0{,}125$.
\end{enumerate}
\end{solucao}
\end{questao}
