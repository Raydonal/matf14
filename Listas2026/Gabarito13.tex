\cabecalho{Gabarito 13: Simulado}{Revisão geral, no formato e no nível da Prova 2: Probabilidade e VA discretas (Cap. 2--3, Aulas 09--23)}

\begin{questao}{1} \textbf{(VA discreta)}
\begin{solucao}
\begin{enumerate}[label=(\alph*)]
  \item $|X-3|\ge2 \Leftrightarrow X\le1 \text{ ou } X\ge5$. $P(X=1)+P(X=5) = 2/10+2/10=4/10=0{,}4$.
  \item $E(X) = 1(0{,}2)+2(0{,}1)+3(0{,}3)+4(0{,}2)+5(0{,}2) = 0{,}2+0{,}2+0{,}9+0{,}8+1{,}0=3{,}1$.
    $E(X^2) = 1(0{,}2)+4(0{,}1)+9(0{,}3)+16(0{,}2)+25(0{,}2) = 0{,}2+0{,}4+2{,}7+3{,}2+5{,}0=11{,}5$.
    $\mathrm{Var}(X) = 11{,}5-3{,}1^2 = 11{,}5-9{,}61=1{,}89$.
  \item $F(1)=0{,}2$; $F(2)=0{,}3$; $F(3)=0{,}6$; $F(4)=0{,}8$; $F(5)=1{,}0$.
\end{enumerate}
\end{solucao}
\end{questao}

\begin{questao}{2} \textbf{(Binomial e aproximação Poisson)}
\begin{solucao}
$X\sim\text{Bin}(150;\,0{,}02)$.
\begin{enumerate}[label=(\alph*)]
  \item $P(X\le3) = \sum_{k=0}^3 \binom{150}{k}(0{,}02)^k(0{,}98)^{150-k} \approx 0{,}6472$.
  \item $\lambda=150\times0{,}02=3$. $P(X\le3)\approx\sum_{k=0}^3 \frac{e^{-3}3^k}{k!} \approx
    0{,}6472$. (Os valores coincidem, ou ficam muito próximos, até a 3ª/4ª casa decimal, a
    aproximação é excelente para $n$ grande e $p$ pequeno.)
\end{enumerate}
\end{solucao}
\end{questao}

\begin{questao}{3} \textbf{(Renda e inadimplência)}
\begin{solucao}
\begin{enumerate}[label=(\alph*)]
  \item $P(\text{inad}) = 100/600 \approx 0{,}167$.
  \item $P(\text{inad}\mid\text{renda baixa}) = 60/240 = 0{,}25$.
  \item Não são independentes: $P(\text{inad}\mid\text{renda baixa})=0{,}25$ é bem diferente de
    $P(\text{inad})\approx0{,}167$ geral (e de $P(\text{inad}\mid\text{renda alta})=40/360\approx
    0{,}111$), a faixa de renda carrega informação relevante sobre a probabilidade de
    inadimplência nesta amostra.
\end{enumerate}
\end{solucao}
\end{questao}

\begin{questao}{4} \textbf{(Exame laboratorial: Bayes)}
\begin{solucao}
$P(\text{doente})=0{,}01$, $P(+\mid\text{doente})=0{,}95$, $P(+\mid\text{sadio})=1-0{,}92=0{,}08$.
\begin{enumerate}[label=(\alph*)]
  \item $P(+) = 0{,}95\times0{,}01 + 0{,}08\times0{,}99 = 0{,}0095+0{,}0792=0{,}0887$.
  \item $P(\text{doente}\mid+) = \dfrac{0{,}0095}{0{,}0887} \approx 0{,}107$ (10,7\%). Mesmo com um
    exame de boa sensibilidade e especificidade, a maioria dos testes positivos ainda é falso
    positivo, porque a doença é rara na população testada, o mesmo raciocínio de base rate visto
    no Cap. 2, Aula 14.
\end{enumerate}
\end{solucao}
\end{questao}
